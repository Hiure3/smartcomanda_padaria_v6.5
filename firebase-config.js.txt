/**
 * 1. MONITORAR MESA (Usado no Painel do Caixa)
 * Fica ouvindo em tempo real tudo o que acontece na mesa (status, pedidos, conexão).
 */
export function monitorarMesa(mesaId, callback) {
  const mesaRef = ref(db, `mesas/mesa_${mesaId}`);
  onValue(mesaRef, (snapshot) => {
    const dados = snapshot.val();
    callback(dados);
  }, (error) => {
    console.error("Erro ao conectar no Realtime Database:", error);
  });
}

/**
 * 2. ATUALIZAR MESA PELO CAIXA (Mensagens ou status)
 */
export function atualizarMesaCaixa(mesaId, dados) {
  const mesaRef = ref(db, `mesas/mesa_${mesaId}`);
  update(mesaRef, dados);
}

/**
 * 3. CLIENTE CONECTAR VIA QR CODE (Usado na página mobile do cliente)
 */
export function registrarConexaoCliente(mesaId) {
  const mesaRef = ref(db, `mesas/mesa_${mesaId}`);
  update(mesaRef, {
    status: "Ocupada",
    clienteOnline: true,
    ultimaConexao: Date.now()
  });
}

/**
 * 4. CLIENTE ENVIAR NOVO PEDIDO (Usado na página mobile do cliente)
 */
export function enviarPedidoCliente(mesaId, nomeItem, preco) {
  const pedidosRef = ref(db, `mesas/mesa_${mesaId}/pedidos`);
  push(pedidosRef, {
    item: nomeItem,
    preco: preco,
    status: "Pronto",
    timestamp: Date.now()
  });
}
